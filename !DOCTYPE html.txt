  <div class="quote-box">
    <h2>Add a Quote/Message</h2>
    <textarea id="quoteInput" placeholder="Write something sweet..."></textarea>
    <button onclick="addQuote()">Save Quote</button>
    <div id="quotesList"></div>
  </div>

  <div class="reminder-box">
    <h2>Add a Reminder</h2>
    <input type="date" id="reminderDate">
    <input type="text" id="reminderText" placeholder="Reminder text...">
    <button onclick="addReminder()">Save Reminder</button>
    <div id="remindersList"></div>
  </div>

  <script>
    function addQuote() {
      let text = document.getElementById("quoteInput").value;
      if(text) {
        let div = document.createElement("p");
        div.textContent = "💌 " + text;
        document.getElementById("quotesList").appendChild(div);
        document.getElementById("quoteInput").value = "";
      }
    }

    function addReminder() {
      let date = document.getElementById("reminderDate").value;
      let text = document.getElementById("reminderText").value;
      if(date && text) {
        let div = document.createElement("p");
        div.textContent = "📅 " + date + " - " + text;
        document.getElementById("remindersList").appendChild(div);
        document.getElementById("reminderDate").value = "";
        document.getElementById("reminderText").value = "";
      }
    }
  </script>
